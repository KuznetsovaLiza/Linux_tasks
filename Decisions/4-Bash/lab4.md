## Задание 1

1. Что такое шебанг?

    Шебанг (shebang, hashbang) — это специальная последовательность символов #! в первой строке скрипта, которая указывает системе, какой интерпретатор использовать для выполнения этого скрипта. Примеры: #!/bin/bash — использовать Bash, #!/usr/bin/python3 — использовать Python 3

2. Обязательно ли исполняемый файл дожен иметь соотвествующее расширение?

    Нет, не обязательно. В Linux/Unix расширения файлов не определяют их исполняемость. Файл становится исполняемым при установке соответствующего права доступа с помощью команды chmod +x filename. Система определяет, как исполнять файл, по шебангу в первой строке файла, правам доступа (бит исполнения). Расширение может помочь пользователю понять тип файла.

3. Напишите скрипт который выполнит автоматически действия из блока работы с файлами (не забудьте включить set -euo pipefail для того что бы ваш скрипт было удобнее отлаживать, опишите что включают эти флаги)

    - Установка строгого режима для отладки (et -euo pipefail):
        - set -e: завершить скрипт при любой ошибке (ненулевом коде возврата)
        - set -u: завершить скрипт при использовании неопределенных переменных
        - set -o pipefail: учитывать ошибки в цепочках пайпов (pipelines)

    - Скрипт:

    ```bash
    #!/bin/bash

    set -euo pipefail

    # Функция для вывода разделителя
    print_separator() {
        echo "=========================================="
        echo "$1"
        echo "=========================================="
    }

    # Создаем временную директорию для работы
    WORKDIR="/tmp/test_workdir"
    echo "Рабочая директория: $WORKDIR"

    print_separator "1. Перемещение между директориями"
    echo "Текущая директория: $(pwd)"
    mkdir -p "$WORKDIR"
    cd "$WORKDIR"
    echo "Перешел в: $(pwd)"

    print_separator "2. Список файлов в директории"
    ls -l

    print_separator "3. Список ВСЕХ файлов в директории (включая скрытые)"
    ls -a 

    print_separator "4. Создание папки с подпапками"
    mkdir -p first_folder/second_folder/third_folder
    echo "Создана структура папок:"
    find first_folder -type d

    print_separator "5. Создание файла с содержимым"
    cd first_folder
    echo "Создаю файл file.txt..."
    echo "Это тестовый файл" > file.txt
    echo "Создан для демонстрации работы с файлами" >> file.txt
    echo "Содержит несколько строк текста" >> file.txt
    echo "Линия 1" >> file.txt
    echo "Линия 3" >> file.txt
    echo "Линия 2" >> file.txt
    echo "Линия 5" >> file.txt
    echo "Линия 4" >> file.txt
    echo "Содержимое созданного файла:"
    cat file.txt

    echo "Создаю второй файл для будущего сравнения..."
    echo "Это тестовый файл" > ../reference.txt
    echo "Содержит несколько строк текста" >> ../reference.txt
    echo "Но с другим содержимым" >> ../reference.txt
    echo "Линия A" >> ../reference.txt
    echo "Линия B" >> ../reference.txt

    print_separator "6. Перемещение файла"
    echo "Перемещаю file.txt из first_folder в родительскую директорию..."
    mv file.txt ../
    echo "Содержимое текущей директории (first_folder):"
    ls -l
    echo "Содержимое родительской директории:"
    cd .. && ls -l

    print_separator "7. Копирование файла"
    echo "Копирую file.txt обратно в first_folder..."
    cp file.txt first_folder/
    echo "Проверяем копирование:"
    echo "В родительской директории:"
    ls file.txt
    echo "В first_folder:"
    ls first_folder/file.txt

    print_separator "8. Переименование файла"
    echo "Переименовываю file.txt в new_file.txt..."
    mv file.txt new_file.txt
    echo "Проверяем переименование:"
    ls -l

    print_separator "9. Сравнение содержимого файлов"
    echo "Сравниваю first_folder/file.txt и reference.txt..."
    echo "Различия между файлами:"
    if diff -u first_folder/file.txt reference.txt; then
        echo "Файлы идентичны"
    else
        echo "Файлы имеют различия"
    fi

    print_separator "10. Сортировка содержимого файла"
    echo "Исходное содержимое first_folder/file.txt:"
    cat first_folder/file.txt
    echo ""
    echo "Сортировка по возрастанию:"
    sort first_folder/file.txt
    echo ""
    echo "Сортировка по убыванию:"
    sort -r first_folder/file.txt

    echo "Создаю файлы с отсортированным содержимым..."
    sort first_folder/file.txt > first_folder/sort_new_file1.txt
    sort -r first_folder/file.txt > first_folder/sort_new_file2.txt
    echo "Созданные файлы:"
    ls -l first_folder/*.txt

    print_separator "11. Удаление всех папок и файлов"
    cd /tmp
    echo "Текущая директория: $(pwd)"
    echo "Удаляю test_workdir..."
    rm -rf test_workdir
    echo "Проверяю удаление:"
    if [ -d "test_workdir" ]; then
        echo "Ошибка: директория все еще существует!"
        exit 1
    else
        echo "Директория успешно удалена"
    fi
    ```
